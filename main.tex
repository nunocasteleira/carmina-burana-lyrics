% --- 2. THE UPDATED MACROS ---

% \draweight{start_num}{end_num}{height}{x_nudge}
\newcommand{\draweight}[4]{%
  \begin{tikzpicture}[remember picture, overlay]
    \coordinate (A) at ($(pic cs:v\themusicverse m#1)+(#4,#3)$);
    \coordinate (B) at ($(pic cs:v\themusicverse m#2)+(#4,#3)$);
    \draw[line width=1.5pt] (A) -- (B); % Beam
    \draw[line width=0.5pt] (A) -- ($(A)-(0,8pt)$); % Start Stem
    \draw[line width=0.5pt] (B) -- ($(B)-(0,8pt)$); % End Stem
  \end{tikzpicture}%
}

% \drawsixteen{start_num}{end_num}{height}{x_nudge}
% This draws two beams and a stem for EVERY marker in the range
\newcommand{\drawsixteen}[4]{%
  \begin{tikzpicture}[remember picture, overlay]
    % Draw the two horizontal beams based on first and last in range
    \coordinate (A) at ($(pic cs:v\themusicverse m#1)+(#4,#3)$);
    \coordinate (B) at ($(pic cs:v\themusicverse m#2)+(#4,#3)$);
    \draw[line width=1.6pt] (A) -- (B);
    \draw[line width=1.6pt] ($(A)-(0,4pt)$) -- ($(B)-(0,4pt)$);

    % Draw a vertical stem for every mark in the numerical range
    \foreach \i in {#1,...,#2} {
      \coordinate (S) at ($(pic cs:v\themusicverse m\i)+(#4,#3)$);
      \draw[line width=0.5pt] (S) -- ($(S)-(0,8pt)$);
    }
  \end{tikzpicture}%
}

% 16th + 16th + 8th: \drawXviViii{start}{end}{height}{nudge}
% Assumes a group of 3 markers (e.g., m1, m2, m3)
\newcommand{\drawXviViii}[4]{%  <-- Added [4] here!
  \begin{tikzpicture}[remember picture, overlay]
    \coordinate (A) at ($(pic cs:v\themusicverse m#1)+(#4,#3)$);
    \coordinate (B) at ($(pic cs:v\themusicverse m\the\numexpr#1+1\relax)+(#4,#3)$); % Second note
    \coordinate (C) at ($(pic cs:v\themusicverse m#2)+(#4,#3)$); % Last note

    % Main top beam (8th note level) across all three
    \draw[line width=1.5pt] (A) -- (C);

    % Secondary beam (16th note level) ONLY between first and second
    \draw[line width=1.5pt] ($(A)-(0,4pt)$) -- ($(B)-(0,4pt)$);

    % Three vertical stems
    \foreach \i in {#1,...,#2} {
      \coordinate (S) at ($(pic cs:v\themusicverse m\i)+(#4,#3)$);
      \draw[line width=0.5pt] (S) -- ($(S)-(0,8pt)$);
    }
  \end{tikzpicture}%
}

% 8th + 16th + 16th: \drawViiiXvi{start}{end}{height}{nudge}
\newcommand{\drawViiiXvi}[4]{% <-- Added [4] here too!
  \begin{tikzpicture}[remember picture, overlay]
    \coordinate (A) at ($(pic cs:v\themusicverse m#1)+(#4,#3)$);
    \coordinate (B) at ($(pic cs:v\themusicverse m\the\numexpr#2-1\relax)+(#4,#3)$); % Second to last
    \coordinate (C) at ($(pic cs:v\themusicverse m#2)+(#4,#3)$);

    \draw[line width=1.5pt] (A) -- (C); % Top beam
    \draw[line width=1.5pt] ($(B)-(0,4pt)$) -- ($(C)-(0,4pt)$); % Lower beam at end

    \foreach \i in {#1,...,#2} {
      \coordinate (S) at ($(pic cs:v\themusicverse m\i)+(#4,#3)$);
      \draw[line width=0.5pt] (S) -- ($(S)-(0,8pt)$);
    }
  \end{tikzpicture}%
}

\newcommand{\drawQuarter}[2]{% #1=mark number, #2=height
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, xshift=3pt, yshift=#2] at (pic cs:v\themusicverse m#1) {\musQuarter};
  \end{tikzpicture}%
}

\newcommand{\drawQuarterDown}[2]{% #1=mark number, #2=height (stem-down: notehead up, stem toward text)
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=center, xshift=3pt, yshift=#2] at (pic cs:v\themusicverse m#1) {\rotatebox{180}{\musQuarter}};
  \end{tikzpicture}%
}
