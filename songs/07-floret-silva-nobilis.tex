  \startline| Floret silva nobilis\\
  \startline| floribus et foliis.\\
  \startline| Ubi est antiquus\\
  \startline| meus amicus?\\
  \startline| Hinc equitavit,\\
  \startline| eia, quis me amabit?\\
\end{keepverse}

\begin{keepverse}

  \startline| wa ist min geselle alse lange?\\
  \startline| Der ist geriten hinnen,\\
  \startline| o wi, wer sol mich minnen?\\
\end{keepverse}
