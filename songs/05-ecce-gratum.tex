  \startline| Estas redit,\\

  \startline| et iam sugit\\
  \startline| Ver estatis ubera;\\
  \startline| illi mens est misera,\\
  \startline| qui nec vivit,\\
  \startline| nec lascivit\\
  \startline| sub Estatis dextera.\\
\end{keepverse}

  \startline| simus jussu Cypridis\\
  \startline| gloriantes\\
  \startline| et letantes\\
