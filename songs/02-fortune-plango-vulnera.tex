\begin{song}{title={2. Fortune plango vulnera}}

\begin{keepverse}
  \startline| Fortune plango vulnera\\
  \startline| stillantibus ocellis\\
  \startline| quod sua michi munera\\
  \startline| subtrahit rebellis.\\
  \startline| Verum est, quod legitur,\\
  \startline| fronte capillata,\\
  \startline| sed plerumque sequitur\\
  \startline| occasio calvata.\\
\end{keepverse}

\begin{keepverse}
  \startline| In Fortune solio\\
  \startline| sederam elatus,\\
  \startline| prosperitatis vario\\
  \startline| flora coronatus;\\
  \startline| quicquid enim florui\\
  \startline| felix et beatus,\\
  \startline| nunc a summo corrui\\
  \startline| gloria privatus.\\
\end{keepverse}

\begin{keepverse}
  \startline| Fortune rota volvitur:\\
  \startline| descendo minoratus;\\
  \startline| alter in altum tollitur;\\
  \startline| nimis exaltatus\\
  \startline| rex sedet in vertice\\
  \startline| caveat ruinam!\\
  \startline| nam sub axe legimus\\
  \startline| Hecubam reginam.\\
\end{keepverse}

\end{multicols}
\end{song}
