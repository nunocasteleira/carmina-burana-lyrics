  \startline| stillantibus ocellis\\
  \startline| quod sua michi munera\\
  \startline| subtrahit rebellis.\\
  \startline| \repstart{fpv}Verum est, quod legitur,\\
  \startline| fronte capillata,\\
  \startline| sed plerumque sequitur\\
  \startline| \repend{fpv}occasio calvata.\\
\end{keepverse}

\begin{keepverse}

  \startline| sederam elatus,\\
  \startline| prosperitatis vario\\
  \startline| flora coronatus;\\
  \startline| \repstart{fpv2}quicquid enim florui\\
  \startline| felix et beatus,\\
  \startline| nunc a summo corrui\\
  \startline| \repend{fpv2}gloria privatus.\\
\end{keepverse}
\drawrep{fpv2}{$2\times$}[\dynmark{p}/\dynmark{f}]

  \startline| nimis exaltatus\\
  \startline| \repstart{fpv3}rex sedet in vertice\\
  \startline| caveat ruinam!\\
  \startline| nam sub axe legimus\\
  \startline| \repend{fpv3}Hecubam reginam.\\
