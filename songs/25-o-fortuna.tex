\songsection{VII.~--- Fortuna Imperatrix Mundi}
\begin{song}{title={25. O Fortuna}}

\begin{keepverse}
  \startline| O Fortuna,\\
  \startline| velut luna\\
  \startline| statu variabilis,\\
  \startline| semper crescis\\
  \startline| aut decrescis;\\
  \startline| vita detestabilis\\
  \startline| nunc obdurat\\
  \startline| et tunc curat\\
  \startline| ludo mentis aciem,\\
  \startline| egestatem,\\
  \startline| potestatem\\
  \startline| dissolvit ut glaciem.\\
\end{keepverse}

\begin{keepverse}
  \startline| Sors immanis\\
  \startline| et inanis,\\
  \startline| rota tu volubilis,\\
  \startline| status malus,\\
  \startline| vana salus\\
  \startline| semper dissolubilis,\\
  \startline| obumbrata\\
  \startline| et velata\\
  \startline| michi quoque niteris;\\
  \startline| nunc per ludum\\
  \startline| dorsum nudum\\
  \startline| fero tui sceleris.\\
\end{keepverse}

\begin{keepverse}
  \startline| Sors salutis\\
  \startline| et virtutis\\
  \startline| michi nunc contraria,\\
  \startline| est affectus\\
  \startline| et defectus\\
  \startline| semper in angaria.\\
  \startline| Hac in hora\\
  \startline| sine mora\\
  \startline| corde pulsum tangite;\\
  \startline| quod per sortem\\
  \startline| sternit fortem,\\
  \startline| mecum omnes plangite!\\
\end{keepverse}

\end{multicols}
\end{song}
